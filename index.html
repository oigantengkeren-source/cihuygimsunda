MODUL: room-system.js Siap Integrasi Frontend/UI

📦 Panduan Integrasi room-system.js

Halaman ini hanyalah Menu Simulasi untuk menguji bahwa sistem Host & Join bekerja. Tim Frontend/UI cukup mengambil file room-system.js dari GitHub ini dan memanggil fungsinya di tombol UI game utama: 

<!-- 1. Pasang script di HTML Game Utama -->
<script src="room-system.js"></script>

<script>
  // 2. Inisialisasi Sistem Room
  const roomSystem = new CihuyRoomSystem({
    maxPlayers: 8,
    onRoomUpdated: (room) => {
      // Update tampilan daftar pemain & kode room di UI kalian
      console.log("Kode Room:", room.code, "Pemain:", room.players);
    },
    onGameStarted: (room, initialData) => {
      // Pindah dari layar Lobby ke layar Game kalian
      mulaiGameUtama(room);
    },
    onGameData: (senderId, data) => {
      // Terima posisi / aksi dari pemain lain secara real-time
      updatePosisiTeman(senderId, data);
    },
    onError: (pesanError) => alert(pesanError)
  });

  // 3. Sambungkan ke Tombol UI Kalian:
  // Saat tombol "Host / Buat Room" ditekan:
  await roomSystem.createRoom("Budi");

  // Saat tombol "Join Room" ditekan:
  await roomSystem.joinRoom("K9X2B", "Siti");

  // Saat pemain klik "Ready":
  roomSystem.toggleReady();

  // Saat Host klik "Start Game":
  roomSystem.startGame();

  // Saat di dalam game ingin kirim data (posisi/skor/aksi):
  roomSystem.sendGameData({ x: 120, y: 80, anim: "run" });
</script>
